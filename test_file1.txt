/* 
    Super stress test for the lexer.
    Mix of everything: keywords, identifiers, ints, reals, comments, and bad chars.
*/

/* Global variable declarations */
integer counter, limit123, integerinteger;
real avg_value, real2, pi_approx;
boolean flag, done, isTrue, isFalse;

function main() {
    integer i, j, k;
    real sum, tempReal, x1, x2;
    boolean result, cond;

    /* Initialize variables (some on same line, some spread out) */
    counter = 0;   limit123 = 100;    integerinteger = 5;
    sum = 0.0; avg_value = 0.000; real2 = 3.14159; pi_approx = 3.14;
    flag = false;  done = false; isTrue = true; isFalse = false;

    read counter;   /* read starting counter */
    read limit123;  /* read limit */

    /* 
       Outer while loop: runs while counter < limit123
       Also check weird spacing and comments inside
    */
    while   (   counter   <   limit123   )   {

        write counter;  /* print the current counter */

        // This line intentionally has bad characters for the lexer:
        // They should be returned as Unknown tokens.
        counter = counter + 1 # $ ;

        /* Inner loop to mess with reals */
        i = 0;
        while (i <= 3) {
            tempReal = 1.0;
            x1 = tempReal + 2;
            x2 = x1 + 3.000;
            sum = sum + x2;
            i = i + 1;
        }

        /* Test different relational and equality operators */
        if (sum == 0.0) 
            result = false;
        otherwise
            result = true;
        fi

        /* Nested if with multiple conditions and weird spacing */
        if    (   result    ==   true   ) {
            write sum;
        }
        otherwise
            write avg_value;
        fi

        /* Try calling write/read/return in odd places */
        if (counter != limit123) 
            write counter;
        otherwise
            write limit123;
        fi

        /* Update average each iteration */
        avg_value = sum / (counter + 1);

        /* Some separators and operators in weird combos */
        j = counter+integerinteger-real2; /* int + int - real */
        k = (j + 10) / 2;

        /* Another boolean expression */
        cond = true;
        if (cond == false)
            flag = true;
        otherwise
            flag = false;
        fi

        /* Random use of @ as separator and more unknowns */
        @ result = true;   /* '@' should be a Separator */
        write result;

        /* superLongIdentifierName12345 to test identifier length */
        superLongIdentifierName12345 = 23.0001;

        counter = counter + 1;
    }

    /* Final while: check >= logic using => (your lexer uses "=>" not ">=") */
    while (counter => limit123) {
        /* This loop probably won’t run, just here for tokens */
        write counter;
        counter = counter - 1;
    }

    /* Final if/otherwise/fi with true/false */
    if (done == false)
        done = true;
    otherwise
        done = false;
    fi

    write done;
    return;
}
